```cpp
void xnLayerDataWidget::OnNavWarnItemDoubleClicked(QTableWidgetItem* item)
{
	if (item) {
		int row = item->row();
		QTableWidgetItem* item = ui.tableNavWarn->item(row, 0);
		int index = item->data(Qt::UserRole).toInt();
		if (index >= 0 && index < m_naviWarnList.size()) {
			NaviWarn nw = m_naviWarnList[index];
			size_t c = nw.m_geo.geometries.size();
			if (c > 0) {
				Latlng ll = nw.m_geo.geometries[0].center;
				GetCurrentCanvas()->View(ll.lat, ll.lng);
			}
		}
	}
}
```

双击表格所选 `index` 行，通过列表 `m_naviWarnList` 获取该行的 `NaviWarn` 类型警告数据的经纬度位置，通过 `GetCurrentCanvas()` 显示出来

## NaviWarn 的地理空间实现

NaviWarnGeometry -> NaviWarnDataGeometry -> NaviWarn

ClipperLib： https://www.angusj.com/clipper2/Docs/Overview.htm

```cpp
enum NaviWarnGeoType
{
    NaviWarnPoint = 0,
    NaviWarnLine,
    NaviWarnPolygon
};

struct NaviWarnBound
{
    ClipperLib::cInt  m_latMin;
    ClipperLib::cInt  m_latMax;
    ClipperLib::cInt  m_lngMin;
    ClipperLib::cInt  m_lngMax;
};

// 多边形结构（包含外环和洞）
struct NaviWarnPolygonData
{
    ClipperLib::Path outerRing;              // 外环
    std::vector<ClipperLib::Path> holes;     // 洞（可以有多个）
    NaviWarnBound bound;                     // 边界
    
    NaviWarnPolygonData() {}
};

// 单个几何体（点/线/面）
struct NaviWarnGeometry
{
    NaviWarnGeoType type;                    // 几何类型
    ClipperLib::Path path;                   // 路径（用于点和线）
    NaviWarnPolygonData polygon;             // 多边形（用于面，包含外环和洞）
    NaviWarnBound bound;                     // 边界
    Latlng center;                           // 中心点
    
    NaviWarnGeometry() : type(NaviWarnPoint) {}
};

// NaviWarn 空间数据（支持面、线、点等）
struct NaviWarnDataGeometry
{
    std::vector<NaviWarnGeometry> geometries; // 所有几何体
    int r;                                    // 半径（用于圆）
    
    NaviWarnDataGeometry() : r(0) {}
};

struct NaviWarn {
	...
	NaviWarnDataGeometry m_geo;                 ///< 空间数据(GML)
}
```

管理 `NaviWarn` 统一使用 `NaviWarnDataManager`，`NaviWarn` 不含任何成员函数

```cpp
class NaviWarnDataManager {
    std::vector<NaviWarn> m_naviWarnLst;
    ...
public:
    static NaviWarnDataManager& GetInstance(void);
    ...
    void GetNaviWarnLst(std::vector<NaviWarn>& lst, NaviWarnWay type);
}
```

该管理类为单例模式，以下为刷新UI表格的时候获取数据的调用：

```cpp
void xnLayerDataWidget::RefreshNavWarn()
{
    m_naviWarnList.clear();
    m_naviWarnList = NaviWarnDataManager::GetInstance().GetNaviWarnLst(NaviWarn_All);
    RefreshNavWarnTableOnly();
}
```
